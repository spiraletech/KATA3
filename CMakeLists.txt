cmake_minimum_required(VERSION 3.20)
project(KATA3 VERSION 0.5.1 LANGUAGES CXX)
add_library(kata3_scene src/scene.cpp src/narrative.cpp src/generation.cpp)
add_library(KATA3::Scene ALIAS kata3_scene)
target_compile_features(kata3_scene PUBLIC cxx_std_20)
target_include_directories(kata3_scene PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>)
include(CTest)
if(BUILD_TESTING)
 add_executable(kata3_tests tests/scene_tests.cpp)
 target_link_libraries(kata3_tests PRIVATE KATA3::Scene)
 if(MSVC)
  target_compile_options(kata3_tests PRIVATE /UNDEBUG)
 else()
  target_compile_options(kata3_tests PRIVATE -UNDEBUG)
 endif()
 add_test(NAME kata3.scene COMMAND kata3_tests)

 if(MSVC)
  target_compile_options(kata3_generation_tests PRIVATE /UNDEBUG)

  target_compile_options(kata3_generation_tests PRIVATE -UNDEBUG)
