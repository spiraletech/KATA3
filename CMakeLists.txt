cmake_minimum_required(VERSION 3.20)
project(KATA3 VERSION 0.7.0 LANGUAGES CXX)
add_library(kata3_scene src/scene.cpp src/narrative.cpp src/generation.cpp src/console.cpp)

target_compile_features(kata3_scene PUBLIC cxx_std_20)
target_include_directories(kata3_scene PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>)
include(CTest)
